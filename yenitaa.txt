def roll_promotion():
    # Probabilidades: 1:15%, 2:30%, 3:30%, 4:15%, 5:10%
    rand = random.random() * 100
    if rand < 15:
        return 1
    elif rand < 45:
        return 2
    elif rand < 75:
        return 3
    elif rand < 90:
        return 4
    else:
        return 5

def generate_code(promo_id):
    suffix = "".join(random.choices("0123456789", k=2))
    bases = {
        1: f"DOUBLERRIEX{suffix}",
        2: f"TWINWITCHX{suffix}",
        3: f"THIRTYCHERRIESX{suffix}",
        4: f"FIFTYFIFTYX{suffix}",
        5: f"FREEGHOSTX{suffix}"
    }
    code = bases.get(promo_id, f"PROMOX{suffix}")
    # Asegurar unicidad total
    while code in redeemed_codes:
        suffix = "".join(random.choices("0123456789", k=2))
        code = bases.get(promo_id, f"PROMOX{suffix}")
    return code

async def redeem(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    user_id = user.id

    if not context.args:
        await update.message.reply_text("Por favor, usa el formato correcto: /redeem + [CODE]")
        return

    input_code = context.args[0].strip()

    # Verificar si el código pertenece al usuario según su tirada
    user_data = user_attempts.get(user_id)
    if not user_data or user_data.get("code") != input_code:
        await update.message.reply_text("El código ingresado no es válido o no te pertenece.")
        return

    if input_code in redeemed_codes:
        await update.message.reply_text("Este código ya ha sido canjeado anteriormente.")
        return

    promo_id = user_data["promo"]
    promo_name = PROMO_NAMES[promo_id]
    username_str = f"@{user.username}" if user.username else user.first_name
    purchase_id = "".join(random.choices("0123456789", k=7))
    
    now_dt = datetime.now(TZ_GMT_MINUS_4)
    # Formato ejemplo: 27/10/2026 1:50 p.m (GMT-4)
    hour_12 = now_dt.strftime("%I:%M").lstrip("0")
    ampm = now_dt.strftime("%p").lower().replace("am", "a.m").replace("pm", "p.m")
    date_str = f"{now_dt.strftime('%d/%m/%Y')} {hour_12} {ampm} (GMT-4)"

    claim_data = {
        "promo_name": promo_name,
        "code": input_code,
        "username": username_str,
        "user_id": user_id,
        "purchase_id": purchase_id,
        "date": date_str
    }

    redeemed_codes[input_code] = claim_data
    user_claims[user_id] = claim_data

    final_message = (

        f"༶⠀𝖴𝗌𝗎𝖺𝗋𝗂𝗈: {username_str}\n"
        f"༶⠀𝖨𝖣 𝖽𝖾 𝖼𝗈𝗆𝗉𝗋𝖺: {purchase_id}\n"
        f"༶⠀𝖥𝖾𝖼𝗁𝖺: {date_str}\n\n"
        "ᬊ   𝖯𝖺𝗋𝖺 𝖿𝗂𝗇𝖺𝗅𝗂𝗓𝖺𝗋 𝗍𝗎 𝗉𝗋𝗈𝖼𝖾𝗌𝗈 𝖽𝖾 𝖼𝗈𝗆𝗉𝗋𝖺, 𝖽𝖾𝖻𝖾𝗌 𝗋𝖾𝖾𝗇𝗏𝗂𝖺𝗋𝗅𝖾 \n"

async def info(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    if user.id not in ADMINS:
        return

    if not context.args:
        await update.message.reply_text("Uso correcto: /info @usuario (o ID)")
        return

    target_query = context.args[0].strip()
    found_claim = None

    for claim in user_claims.values():
        if target_query.lower() in claim["username"].lower() or target_query == str(claim["user_id"]):
            found_claim = claim
            break

    if not found_claim:
        await update.message.reply_text("No se encontró ningún registro para ese usuario.")
        return

    info_text = (
        "ㅤㅤㅤㅤㅤㅤ░.⠀۟⠀｡ㅤㅤ𝓤𝗌𝖾𝗋'𝗌 𝗂𝗇𝖿𝗈ㅤㅤㅤㅤㅤㅤㅤㅤㅤㅤ\n\n"
